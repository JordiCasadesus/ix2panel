2. PowerShell como Administrador en el PC Windows:

            
# Instalar OpenSSH Server
Add-WindowsCapability -Online -Name OpenSSH.Server~~~~0.0.1.0
Start-Service sshd
Set-Service -Name sshd -StartupType Automatic

# Pegar la clave publica del NAS (resultado del cat anterior):
$key = "PEGAR_AQUI_LA_CLAVE_PUB"
$f = "$env:ProgramData\ssh\administrators_authorized_keys"
if (!(Test-Path "$env:ProgramData\ssh")) { New-Item -ItemType Directory "$env:ProgramData\ssh" }
Set-Content $f $key -Encoding utf8
icacls $f /inheritance:r
icacls $f /grant "*S-1-5-32-544:F"
icacls $f /grant "*S-1-5-18:F"
New-NetFirewallRule -Name "sshd" -DisplayName "OpenSSH Server" -Protocol TCP -LocalPort 22 -Action Allow -Direction Inbound -ErrorAction SilentlyContinue
Restart-Service sshd
